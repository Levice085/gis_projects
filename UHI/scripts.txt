# Define the region of interest based on Mvita constituency.
mvita = admin.filter(ee.Filter.eq('CONSTITUEN', 'MVITA'))
geometry = mvita.geometry()
Map.centerObject(geometry, 15)

# Function to filter out cloudy pixels.
def cloudMask(cloudyScene):
scored = ee.Algorithms.Landsat.simpleCloudScore(cloudyScene)
mask = scored.select(['cloud']).lte(10)
return cloudyScene.updateMask(mask)

# Generate a water mask.
water = ee.Image('JRC/GSW1_0/GlobalSurfaceWater').select('occurrence')
notWater = water.mask().Not()

# Define the date range
now = ee.Date(Date.now())

# Landsat collection filtering based on cloud cover and date.
filtered = landsat.filter(ee.Filter.lt('CLOUD_COVER', 20)) \ \
.filter(ee.Filter.date('2014-01-01', '2025-01-01')) \ \
.filter(ee.Filter.bounds(geometry)) \ \
.map(cloudMask)

median = filtered.median()
clipped = median.clip(geometry)

rgbVis = {
    "min": 0,
    "max": 1,
    "bands": ['B4', 'B3', 'B2']
}
print('Landsat collection', filtered.getInfo().getInfo())

# Select thermal band 10 (with brightness temperature).
thermal = median.select('B10') \ \
.clip(geometry) \ \
.updateMask(notWater)

Map.addLayer(thermal, {
    "min": 295,
    "max": 310,
    "palette": ['#a50026', '#d73027', '#f46d43', '#fdae61', '#fee090', '#e0f3f8', '#abd9e9', '#74add1', '#4575b4', '#313695']
},
'Landsat_BT')

# Function to apply scaling factor for BT conversion
def applyScaleFactors(image):
bt = image.select('B10').multiply(0.00341802).add(149.0)
return bt.rename('BT')

# Apply scaling factor to all images
btCollection = filtered.map(applyScaleFactors)

# Compute mean BT over time
meanBT = btCollection.mean()

# Calculate Normalized Difference Vegetation Index (NDVI)
dataset = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2') \ \
.filterBounds(geometry) \ \
.filterDate('2014-01-01', '2025-01-01')

def func_foe(image):
ndvi = image.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDVI')
ndbi = image.normalizedDifference(['SR_B6','SR_B5']).rename(['NDBI'])
ndvi.updateMask(notWater)
ndbi.updateMask(notWater)
return image.addBands(ndvi).addBands(ndbi)

computeNdvi = func_foe

withNdvi = dataset.map(computeNdvi)
compositeNdvi = withNdvi.median().clip(geometry).select('NDVI')
compositeNdbi = withNdvi.median().clip(geometry).select('NDBI')

Map.addLayer(compositeNdvi, {
    "min": 0,
    "max": 0.7,
    "palette": ['FFFFFF', 'CE7E45', 'DF923D', 'F1B555', 'FCD163', '99B718', '74A901', '66A000', '529400', '3E8601', '207401', '056201', '004C00', '023B01', '012E01', '011D01', '011301']
}, 'NDVI')

ndbiVis = {
    "min":0,
    "max":0.7,
    "palette":['white','red']
}
Map.addLayer(compositeNdbi,ndbiVis,'NDBI')

#Exporting the NDVI for visualization
Export.image.toDrive(

#Exporting the NDBI for visualization

# Find the minimum and maximum of NDVI.
minMax = compositeNdvi.reduceRegion(
"reducer"=ee.Reducer.min().combine(
"reducer2"=ee.Reducer.max()
"sharedInputs"=True
)
"geometry"=geometry
"scale"=30
"maxPixels"=1e9
)
print('minMax', minMax.getInfo().getInfo())

min = ee.Number(minMax.get('NDVI_min'))
max = ee.Number(minMax.get('NDVI_max'))

# Calculate fractional vegetation (FV).
fv = compositeNdvi.subtract(min).divide(max.subtract(min)).rename('FV')

# Add FV to the image collection

withfv = compositeNdvi.addBands(fv).addBands(compositeNdbi)
Map.addLayer(fv, {
    "min": 0,
    "max": 0.7,
    "palette": ['FFFFFF', 'CE7E45', 'DF923D', 'F1B555', 'FCD163', '99B718', '74A901', '66A000', '529400', '3E8601', '207401', '056201', '004C00', '023B01', '012E01', '011D01', '011301']
}, 'FV')

# Export FV image

"folder"='FV'
"fileNamePrefix"='fv'
"region"=geometry
"scale"=10
"maxPixels"=1e9
)

# Emissivity calculations.
a = ee.Number(0.004)
b = ee.Number(0.986)
em = fv.multiply(a).add(b).rename('EMM').updateMask(notWater)

Map.addLayer(em, {
    "min": 0.98,
    "max": 0.99,

"folder"='EMM'

"region"=geometry
"scale"=10
"maxPixels"=1e9
)

# Calculate LST from emissivity and brightness temperature.
lstLandsat = thermal.expression(
'(Tb / (1 + (0.001145 * (Tb / 1.438)) * log(Ep))) - 273.15', {
    'Tb': thermal.select('B10'),
    'Ep': em.select('EMM')
}).updateMask(notWater).rename('LST')

Map.addLayer(lstLandsat, {
    "min": 25,
    "max": 35,

}, 'LST_Landsat')

# Convert SUHI to Integer Classes for Vectorization
suhiClassified = suhi.expression(
"(b(0) < 2) ? 1" + \
": (b(0) < 5) ? 2" + \
": 3",
{
    "b(0)": suhi
}).rename("SUHI_Class").toByte()

# Remove the zero values
suhiMasked = suhiClassified.selfMask()

# Add an extra column to avoid the error
suhiForVector = suhiMasked.addBands(ee.Image.constant(1).rename("countBand"))

# Convert SUHI Raster to Vector
suhiVector = suhiForVector.reduceToVectors(
"geometry"=geometry
"scale"=30
"geometryType"="polygon"
"reducer"=ee.Reducer.count()
"maxPixels"=1e13
)
